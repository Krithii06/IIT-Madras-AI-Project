# Runtime dependencies for the Vercel serverless functions in api/.
# Deliberately the same pins as backend/requirements.txt minus the web framework:
# Vercel provides the HTTP layer, so FastAPI and uvicorn are not needed here.
#

# and the build failed at dependency resolution. Bump both together.

numpy==2.2.4
